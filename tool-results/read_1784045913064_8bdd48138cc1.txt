   251→   251→function ResourcesSection() {
   252→   252→  const [resources, setResources] = useState<Resource[]>([]);
   253→   253→  const [categories, setCategories] = useState<string[]>([]);
   254→   254→  const [search, setSearch] = useState("");
   255→   255→  const [activeCategory, setActiveCategory] = useState("All");
   256→   256→  const [showFavorites, setShowFavorites] = useState(false);
   257→   257→  const [loading, setLoading] = useState(true);
   258→   258→  const [dialogOpen, setDialogOpen] = useState(false);
   259→   259→  const [editItem, setEditItem] = useState<Resource | null>(null);
   260→   260→  const [formTitle, setFormTitle] = useState("");
   261→   261→  const [formUrl, setFormUrl] = useState("");
   262→   262→  const [formDesc, setFormDesc] = useState("");
   263→   263→  const [formCategory, setFormCategory] = useState("General");
   264→   264→  const [formTags, setFormTags] = useState("");
   265→   265→
   266→   266→  const fetchResources = useCallback(async () => {
   267→   267→    try {
   268→   268→      setLoading(true);
   269→   269→      const params = new URLSearchParams();
   270→   270→      if (activeCategory !== "All") params.set("category", activeCategory);
   271→   271→      if (search) params.set("search", search);
   272→   272→      if (showFavorites) params.set("favorites", "true");
   273→   273→
   274→   274→      const res = await fetch(`/api/resources?${params}`);
   275→   275→      const data = await res.json();
   276→   276→      setResources(data.resources || []);
   277→   277→      setCategories(data.categories || []);
   278→   278→    } catch {
   279→   279→      toast.error("Failed to load resources");
   280→   280→    } finally {
   281→   281→      setLoading(false);
   282→   282→    }
   283→   283→  }, [activeCategory, search, showFavorites]);
   284→   284→
   285→   285→  useEffect(() => {
   286→   286→    fetchResources();
   287→   287→  }, [fetchResources]);
   288→   288→
   289→   289→  const handleSubmit = async () => {
   290→   290→    if (!formTitle.trim() || !formUrl.trim()) {
   291→   291→      toast.error("Title and URL are required");
   292→   292→      return;
   293→   293→    }
   294→   294→
   295→   295→    try {
   296→   296→      let url = formUrl.trim();
   297→   297→      if (!url.startsWith("http")) url = `https://${url}`;
   298→   298→
   299→   299→      const body = {
   300→   300→        title: formTitle.trim(),
   301→   301→        url,
   302→   302→        description: formDesc.trim(),
   303→   303→        category: formCategory,
   304→   304→        tags: formTags.trim(),
   305→   305→        faviconUrl: `https://www.google.com/s2/favicons?domain=${new URL(url).hostname}&sz=64`,
   306→   306→      };
   307→   307→
   308→   308→      if (editItem) {
   309→   309→        await fetch("/api/resources", {
   310→   310→          method: "PUT",
   311→   311→          headers: { "Content-Type": "application/json" },
   312→   312→          body: JSON.stringify({ id: editItem.id, ...body }),
   313→   313→        });
   314→   314→        toast.success("Resource updated");
   315→   315→      } else {
   316→   316→        await fetch("/api/resources", {
   317→   317→          method: "POST",
   318→   318→          headers: { "Content-Type": "application/json" },
   319→   319→          body: JSON.stringify(body),
   320→   320→        });
   321→   321→        toast.success("Resource added");
   322→   322→      }
   323→   323→
   324→   324→      setDialogOpen(false);
   325→   325→      resetForm();
   326→   326→      fetchResources();
   327→   327→    } catch {
   328→   328→      toast.error("Failed to save resource");
   329→   329→    }
   330→   330→  };
   331→   331→
   332→   332→  const handleDelete = async (id: string) => {
   333→   333→    try {
   334→   334→      await fetch(`/api/resources?id=${id}`, { method: "DELETE" });
   335→   335→      toast.success("Resource deleted");
   336→   336→      fetchResources();
   337→   337→    } catch {
   338→   338→      toast.error("Failed to delete");
   339→   339→    }
   340→   340→  };
   341→   341→
   342→   342→  const toggleFavorite = async (r: Resource) => {
   343→   343→    try {
   344→   344→      await fetch("/api/resources", {
   345→   345→        method: "PUT",
   346→   346→        headers: { "Content-Type": "application/json" },
   347→   347→        body: JSON.stringify({ id: r.id, isFavorite: !r.isFavorite }),
   348→   348→      });
   349→   349→      fetchResources();
   350→   350→    } catch {
   351→   351→      toast.error("Failed to update");
   352→   352→    }
   353→   353→  };
   354→   354→
   355→   355→  const openEdit = (r: Resource) => {
   356→   356→    setEditItem(r);
   357→   357→    setFormTitle(r.title);
   358→   358→    setFormUrl(r.url);
   359→   359→    setFormDesc(r.description || "");
   360→   360→    setFormCategory(r.category);
   361→   361→    setFormTags(r.tags);
   362→   362→    setDialogOpen(true);
   363→   363→  };
   364→   364→
   365→   365→  const openAdd = () => {
   366→   366→    setEditItem(null);
   367→   367→    resetForm();
   368→   368→    setDialogOpen(true);
   369→   369→  };
   370→   370→
   371→   371→  const resetForm = () => {
   372→   372→    setEditItem(null);
   373→   373→    setFormTitle("");
   374→   374→    setFormUrl("");
   375→   375→    setFormDesc("");
   376→   376→    setFormCategory("General");
   377→   377→    setFormTags("");
   378→   378→  };
   379→   379→
   380→   380→  const allCategories = ["All", ...categories];
   381→   381→
   382→   382→  return (
   383→   383→    <div className="max-w-[1400px] mx-auto px-4 sm:px-6 py-6">
   384→   384→      {/* TOP BAR */}
   385→   385→      <div className="flex flex-col sm:flex-row items-start sm:items-center justify-between gap-4 mb-6">
   386→   386→        <div>
   387→   387→          <h2 className="text-2xl font-bold tracking-tight">Resource Library</h2>
   388→   388→          <p className="text-sm text-muted-foreground mt-1">
   389→   389→            {resources.length} resources in your collection
   390→   390→          </p>
   391→   391→        </div>
   392→   392→        <Button onClick={openAdd} className="gap-2">
   393→   393→          <Plus className="w-4 h-4" />
   394→   394→          Add Resource
   395→   395→        </Button>
   396→   396→      </div>
   397→   397→
   398→   398→      {/* SEARCH & FILTERS */}
   399→   399→      <div className="flex flex-col sm:flex-row gap-3 mb-6">
   400→   400→        <div className="relative flex-1">
   401→   401→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   402→   402→          <Input
   403→   403→            placeholder="Search resources..."
   404→   404→            value={search}
   405→   405→            onChange={(e) => setSearch(e.target.value)}
   406→   406→            className="pl-9 h-10"
   407→   407→          />
   408→   408→        </div>
   409→   409→        <div className="flex gap-2">
   410→   410→          <Button
   411→   411→            variant={showFavorites ? "default" : "outline"}
   412→   412→            size="sm"
   413→   413→            onClick={() => setShowFavorites(!showFavorites)}
   414→   414→            className="gap-1.5 h-10"
   415→   415→          >
   416→   416→            <Heart className={`w-3.5 h-3.5 ${showFavorites ? "fill-current" : ""}`} />
   417→   417→            Favorites
   418→   418→          </Button>
   419→   419→        </div>
   420→   420→      </div>
   421→   421→
   422→   422→      {/* CATEGORY FILTERS */}
   423→   423→      <div className="flex gap-2 mb-6 overflow-x-auto pb-1 scrollbar-none">
   424→   424→        {allCategories.map((cat) => (
   425→   425→          <button
   426→   426→            key={cat}
   427→   427→            onClick={() => setActiveCategory(cat)}
   428→   428→            className={`inline-flex items-center gap-1.5 px-3 py-1.5 rounded-full text-sm font-medium whitespace-nowrap transition-colors ${
   429→   429→              activeCategory === cat
   430→   430→                ? "bg-foreground text-background"
   431→   431→                : "bg-secondary text-secondary-foreground hover:bg-secondary/80"
   432→   432→            }`}
   433→   433→          >
   434→   434→            {cat !== "All" && <span className="text-xs">{getCategoryIcon(cat)}</span>}
   435→   435→            {cat}
   436→   436→          </button>
   437→   437→        ))}
   438→   438→      </div>
   439→   439→
   440→   440→      {/* GRID */}
   441→   441→      {loading ? (
   442→   442→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   443→   443→          {Array.from({ length: 6 }).map((_, i) => (
   444→   444→            <Card key={i} className="overflow-hidden">
   445→   445→              <CardContent className="p-5">
   446→   446→                <div className="flex items-start gap-3">
   447→   447→                  <Skeleton className="w-10 h-10 rounded-lg" />
   448→   448→                  <div className="flex-1">
   449→   449→                    <Skeleton className="h-5 w-24 mb-2" />
   450→   450→                    <Skeleton className="h-3 w-full mb-1" />
   451→   451→                    <Skeleton className="h-3 w-3/4" />
   452→   452→                  </div>
   453→   453→                </div>
   454→   454→              </CardContent>
   455→   455→            </Card>
   456→   456→          ))}
   457→   457→        </div>
   458→   458→      ) : resources.length === 0 ? (
   459→   459→        <div className="text-center py-20">
   460→   460→          <FolderOpen className="w-12 h-12 mx-auto text-muted-foreground/50 mb-4" />
   461→   461→          <h3 className="text-lg font-medium mb-1">No resources found</h3>
   462→   462→          <p className="text-sm text-muted-foreground">
   463→   463→            {showFavorites
   464→   464→              ? "You haven't favorited any resources yet"
   465→   465→              : "Try adjusting your search or filters"}
   466→   466→          </p>
   467→   467→        </div>
   468→   468→      ) : (
   469→   469→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   470→   470→          {resources.map((r) => (
   471→   471→            <ResourceCard
   472→   472→              key={r.id}
   473→   473→              resource={r}
   474→   474→              onFavorite={() => toggleFavorite(r)}
   475→   475→              onEdit={() => openEdit(r)}
   476→   476→              onDelete={() => handleDelete(r.id)}
   477→   477→            />
   478→   478→          ))}
   479→   479→        </div>

   212→function ResourcesSection() {
   213→  const [resources, setResources] = useState<Resource[]>([]);
   214→  const [categories, setCategories] = useState<string[]>([]);
   215→  const [search, setSearch] = useState("");
   216→  const [activeCategory, setActiveCategory] = useState("All");
   217→  const [showFavorites, setShowFavorites] = useState(false);
   218→  const [loading, setLoading] = useState(true);
   219→  const [dialogOpen, setDialogOpen] = useState(false);
   220→  const [editItem, setEditItem] = useState<Resource | null>(null);
   221→  const [formTitle, setFormTitle] = useState("");
   222→  const [formUrl, setFormUrl] = useState("");
   223→  const [formDesc, setFormDesc] = useState("");
   224→  const [formCategory, setFormCategory] = useState("General");
   225→  const [formTags, setFormTags] = useState("");
   226→
   227→  const fetchResources = useCallback(async () => {
   228→    try {
   229→      setLoading(true);
   230→      const params = new URLSearchParams();
   231→      if (activeCategory !== "All") params.set("category", activeCategory);
   232→      if (search) params.set("search", search);
   233→      if (showFavorites) params.set("favorites", "true");
   234→
   235→      const res = await fetch(`/api/resources?${params}`);
   236→      const data = await res.json();
   237→      setResources(data.resources || []);
   238→      setCategories(data.categories || []);
   239→    } catch {
   240→      toast.error("Failed to load resources");
   241→    } finally {
   242→      setLoading(false);
   243→    }
   244→  }, [activeCategory, search, showFavorites]);
   245→
   246→  useEffect(() => {
   247→    fetchResources();
   248→  }, [fetchResources]);
   249→
   250→  const handleSubmit = async () => {
   251→    if (!formTitle.trim() || !formUrl.trim()) {
   252→      toast.error("Title and URL are required");
   253→      return;
   254→    }
   255→
   256→    try {
   257→      let url = formUrl.trim();
   258→      if (!url.startsWith("http")) url = `https://${url}`;
   259→
   260→      const body = {
   261→        title: formTitle.trim(),
   262→        url,
   263→        description: formDesc.trim(),
   264→        category: formCategory,
   265→        tags: formTags.trim(),
   266→        faviconUrl: `https://www.google.com/s2/favicons?domain=${new URL(url).hostname}&sz=64`,
   267→      };
   268→
   269→      if (editItem) {
   270→        await fetch("/api/resources", {
   271→          method: "PUT",
   272→          headers: { "Content-Type": "application/json" },
   273→          body: JSON.stringify({ id: editItem.id, ...body }),
   274→        });
   275→        toast.success("Resource updated");
   276→      } else {
   277→        await fetch("/api/resources", {
   278→          method: "POST",
   279→          headers: { "Content-Type": "application/json" },
   280→          body: JSON.stringify(body),
   281→        });
   282→        toast.success("Resource added");
   283→      }
   284→
   285→      setDialogOpen(false);
   286→      resetForm();
   287→      fetchResources();
   288→    } catch {
   289→      toast.error("Failed to save resource");
   290→    }
   291→  };
   292→
   293→  const handleDelete = async (id: string) => {
   294→    try {
   295→      await fetch(`/api/resources?id=${id}`, { method: "DELETE" });
   296→      toast.success("Resource deleted");
   297→      fetchResources();
   298→    } catch {
   299→      toast.error("Failed to delete");
   300→    }
   301→  };
   302→
   303→  const toggleFavorite = async (r: Resource) => {
   304→    try {
   305→      await fetch("/api/resources", {
   306→        method: "PUT",
   307→        headers: { "Content-Type": "application/json" },
   308→        body: JSON.stringify({ id: r.id, isFavorite: !r.isFavorite }),
   309→      });
   310→      fetchResources();
   311→    } catch {
   312→      toast.error("Failed to update");
   313→    }
   314→  };
   315→
   316→  const openEdit = (r: Resource) => {
   317→    setEditItem(r);
   318→    setFormTitle(r.title);
   319→    setFormUrl(r.url);
   320→    setFormDesc(r.description || "");
   321→    setFormCategory(r.category);
   322→    setFormTags(r.tags);
   323→    setDialogOpen(true);
   324→  };
   325→
   326→  const openAdd = () => {
   327→    setEditItem(null);
   328→    resetForm();
   329→    setDialogOpen(true);
   330→  };
   331→
   332→  const resetForm = () => {
   333→    setEditItem(null);
   334→    setFormTitle("");
   335→    setFormUrl("");
   336→    setFormDesc("");
   337→    setFormCategory("General");
   338→    setFormTags("");
   339→  };
   340→
   341→  const allCategories = ["All", ...categories];
   342→
   343→  return (
   344→    <div className="max-w-[1200px] mx-auto px-6 py-8">
   345→      {/* TOP BAR */}
   346→      <div className="flex flex-col sm:flex-row items-start sm:items-center justify-between gap-4 mb-8">
   347→        <div>
   348→          <h2 className="text-[24px] font-bold tracking-tight text-white" style={{ lineHeight: 1.2 }}>Resource Library</h2>
   349→          <p className="text-[14px] text-[#A0A0A0] mt-1" style={{ lineHeight: 1.5 }}>
   350→            {resources.length} resources in your collection
   351→          </p>
   352→        </div>
   353→        <button
   354→          onClick={openAdd}
   355→          className="inline-flex items-center gap-2 px-5 py-3 bg-[#FF6B35] hover:bg-[#FF5722] active:bg-[#E64A19] text-white rounded-lg text-[14px] font-semibold transition-all duration-200"
   356→        >
   357→          <Plus className="w-4 h-4" />
   358→          Add Resource
   359→        </button>
   360→      </div>
   361→
   362→      {/* SEARCH & FILTERS */}
   363→      <div className="flex flex-col sm:flex-row gap-3 mb-6">
   364→        <div className="relative flex-1">
   365→          <Search className="absolute left-4 top-1/2 -translate-y-1/2 w-5 h-5 text-[#606060]" />
   366→          <input
   367→            placeholder="Search resources..."
   368→            value={search}
   369→            onChange={(e) => setSearch(e.target.value)}
   370→            className="w-full pl-12 pr-4 py-3 bg-[#2A2A2A] border border-[#333333] rounded-lg text-[14px] text-white placeholder:text-[#606060] outline-none focus:border-[#FF6B35] transition-colors duration-200"
   371→          />
   372→        </div>
   373→        <button
   374→          variant={showFavorites ? "default" : "outline"}
   375→          onClick={() => setShowFavorites(!showFavorites)}
   376→          className={`inline-flex items-center gap-1.5 px-4 py-3 rounded-lg text-[14px] font-medium transition-all duration-200 ${
   377→            showFavorites
   378→              ? "bg-[#FF6B35] text-white"
   379→              : "bg-[#333333] text-[#A0A0A0] hover:bg-[#444444] hover:text-white"
   380→          }`}
   381→        >
   382→          <Heart className={`w-4 h-4 ${showFavorites ? "fill-current" : ""}`} />
   383→          Favorites
   384→        </button>
   385→      </div>
   386→
   387→      {/* CATEGORY FILTERS */}
   388→      <div className="flex gap-2 mb-6 overflow-x-auto pb-1 scrollbar-none">
   389→        {allCategories.map((cat) => (
   390→          <button
   391→            key={cat}
   392→            onClick={() => setActiveCategory(cat)}
   393→            className={`inline-flex items-center gap-1.5 px-4 py-2 rounded-full text-[12px] font-medium whitespace-nowrap transition-all duration-200 ${
   394→              activeCategory === cat
   395→                ? "bg-[#FF6B35] text-white"
   396→                : "bg-[#333333] text-[#A0A0A0] hover:bg-[#444444] hover:text-white"
   397→            }`}
   398→          >
   399→            {cat !== "All" && <span className="text-xs">{getCategoryIcon(cat)}</span>}
   400→            {cat}
   401→          </button>
   402→        ))}
   403→      </div>
   404→
   405→      {/* GRID */}
   406→      {loading ? (
   407→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-6">
   408→          {Array.from({ length: 6 }).map((_, i) => (
   409→            <div key={i} className="bg-[#1E1E1E] border border-[#333333] rounded-xl p-5">
   410→              <div className="flex items-start gap-3">
   411→                <Skeleton className="w-10 h-10 rounded-lg bg-[#333333]" />
   412→                <div className="flex-1">
   413→                  <Skeleton className="h-5 w-24 mb-2 bg-[#333333]" />
   414→                  <Skeleton className="h-3 w-full mb-1 bg-[#333333]" />
   415→                  <Skeleton className="h-3 w-3/4 bg-[#333333]" />
   416→                </div>
   417→              </div>
   418→            </div>
   419→          ))}
   420→        </div>
   421→      ) : resources.length === 0 ? (
   422→        <div className="text-center py-20">
   423→          <FolderOpen className="w-12 h-12 mx-auto text-[#606060] mb-4" />
   424→          <h3 className="text-[18px] font-semibold text-white mb-1">No resources found</h3>
   425→          <p className="text-[14px] text-[#A0A0A0]">
   426→            {showFavorites
   427→              ? "You haven't favorited any resources yet"
   428→              : "Try adjusting your search or filters"}
   429→          </p>
   430→        </div>
   431→      ) : (
   432→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-6">
   433→          {resources.map((r) => (
   434→            <ResourceCard
   435→              key={r.id}
   436→              resource={r}
   437→              onFavorite={() => toggleFavorite(r)}
   438→              onEdit={() => openEdit(r)}
   439→              onDelete={() => handleDelete(r.id)}
   440→            />
   441→          ))}
   442→        </div>
   443→      )}
   444→
   445→      {/* ADD/EDIT DIALOG */}
   446→      <Dialog open={dialogOpen} onOpenChange={(o) => { setDialogOpen(o); if (!o) resetForm(); }}>
   447→        <DialogContent className="sm:max-w-[480px] bg-[#1E1E1E] border-[#333333] rounded-xl">
   448→          <DialogHeader>
   449→            <DialogTitle className="text-white">{editItem ? "Edit Resource" : "Add Resource"}</DialogTitle>
   450→          </DialogHeader>
   451→          <div className="space-y-4 mt-2">
   452→            <div className="space-y-2">
   453→              <label className="text-[14px] font-medium text-[#A0A0A0]">Title *</label>
   454→              <input
   455→                placeholder="e.g. Figma"
   456→                value={formTitle}
   457→                onChange={(e) => setFormTitle(e.target.value)}
   458→                className="w-full px-4 py-3 bg-[#2A2A2A] border border-[#333333] rounded-lg text-[14px] text-white placeholder:text-[#606060] outline-none focus:border-[#FF6B35] transition-colors duration-200"
   459→              />
   460→            </div>
   461→            <div className="space-y-2">
   462→              <label className="text-[14px] font-medium text-[#A0A0A0]">URL *</label>

   478→              />
   479→            </div>
   480→            <div className="space-y-2">
   481→              <label className="text-[14px] font-medium text-[#A0A0A0]">Category</label>

   487→                  {["Design", "Assets", "Development", "AI", "Productivity", "Reference", "Typography", "General"].map(

   528→
   529→function ResourceCard({
   530→  resource,
   531→  onFavorite,
   532→  onEdit,
   533→  onDelete,
   534→}: {
   535→  resource: Resource;
   536→  onFavorite: () => void;
   537→  onEdit: () => void;
   538→  onDelete: () => void;
   539→}) {
   540→  const [showMenu, setShowMenu] = useState(false);
   541→  const menuRef = useRef<HTMLDivElement>(null);
   542→  const hostname = (() => {
   543→    try { return new URL(resource.url).hostname; } catch { return resource.url; }
   544→  })();
   545→
   546→  useEffect(() => {
   547→    const handleClickOutside = (e: MouseEvent) => {
   548→      if (menuRef.current && !menuRef.current.contains(e.target as Node)) {
   549→        setShowMenu(false);
   550→      }
   551→    };
   552→    if (showMenu) document.addEventListener("mousedown", handleClickOutside);
   553→    return () => document.removeEventListener("mousedown", handleClickOutside);
   554→  }, [showMenu]);
   555→
   556→  return (
   557→    <motion.div
   558→      layout
   559→      initial={{ opacity: 0, y: 12 }}
   560→      animate={{ opacity: 1, y: 0 }}

   251→function ResourcesSection() {
   252→  const [resources, setResources] = useState<Resource[]>([]);
   253→  const [categories, setCategories] = useState<string[]>([]);
   254→  const [search, setSearch] = useState("");
   255→  const [activeCategory, setActiveCategory] = useState("All");
   256→  const [showFavorites, setShowFavorites] = useState(false);
   257→  const [loading, setLoading] = useState(true);
   258→  const [dialogOpen, setDialogOpen] = useState(false);
   259→  const [editItem, setEditItem] = useState<Resource | null>(null);
   260→  const [formTitle, setFormTitle] = useState("");
   261→  const [formUrl, setFormUrl] = useState("");
   262→  const [formDesc, setFormDesc] = useState("");
   263→  const [formCategory, setFormCategory] = useState("General");
   264→  const [formTags, setFormTags] = useState("");
   265→
   266→  const fetchResources = useCallback(async () => {
   267→    try {
   268→      setLoading(true);
   269→      const params = new URLSearchParams();
   270→      if (activeCategory !== "All") params.set("category", activeCategory);
   271→      if (search) params.set("search", search);
   272→      if (showFavorites) params.set("favorites", "true");
   273→
   274→      const res = await fetch(`/api/resources?${params}`);
   275→      const data = await res.json();
   276→      setResources(data.resources || []);
   277→      setCategories(data.categories || []);
   278→    } catch {
   279→      toast.error("Failed to load resources");
   280→    } finally {
   281→      setLoading(false);
   282→    }
   283→  }, [activeCategory, search, showFavorites]);
   284→
   285→  useEffect(() => {
   286→    fetchResources();
   287→  }, [fetchResources]);
   288→
   289→  const handleSubmit = async () => {
   290→    if (!formTitle.trim() || !formUrl.trim()) {
   291→      toast.error("Title and URL are required");
   292→      return;
   293→    }
   294→
   295→    try {
   296→      let url = formUrl.trim();
   297→      if (!url.startsWith("http")) url = `https://${url}`;
   298→
   299→      const body = {
   300→        title: formTitle.trim(),
   301→        url,
   302→        description: formDesc.trim(),
   303→        category: formCategory,
   304→        tags: formTags.trim(),
   305→        faviconUrl: `https://www.google.com/s2/favicons?domain=${new URL(url).hostname}&sz=64`,
   306→      };
   307→
   308→      if (editItem) {
   309→        await fetch("/api/resources", {
   310→          method: "PUT",
   311→          headers: { "Content-Type": "application/json" },
   312→          body: JSON.stringify({ id: editItem.id, ...body }),
   313→        });
   314→        toast.success("Resource updated");
   315→      } else {
   316→        await fetch("/api/resources", {
   317→          method: "POST",
   318→          headers: { "Content-Type": "application/json" },
   319→          body: JSON.stringify(body),
   320→        });
   321→        toast.success("Resource added");
   322→      }
   323→
   324→      setDialogOpen(false);
   325→      resetForm();
   326→      fetchResources();
   327→    } catch {
   328→      toast.error("Failed to save resource");
   329→    }
   330→  };
   331→
   332→  const handleDelete = async (id: string) => {
   333→    try {
   334→      await fetch(`/api/resources?id=${id}`, { method: "DELETE" });
   335→      toast.success("Resource deleted");
   336→      fetchResources();
   337→    } catch {
   338→      toast.error("Failed to delete");
   339→    }
   340→  };
   341→
   342→  const toggleFavorite = async (r: Resource) => {
   343→    try {
   344→      await fetch("/api/resources", {
   345→        method: "PUT",
   346→        headers: { "Content-Type": "application/json" },
   347→        body: JSON.stringify({ id: r.id, isFavorite: !r.isFavorite }),
   348→      });
   349→      fetchResources();
   350→    } catch {
   351→      toast.error("Failed to update");
   352→    }
   353→  };
   354→
   355→  const openEdit = (r: Resource) => {
   356→    setEditItem(r);
   357→    setFormTitle(r.title);
   358→    setFormUrl(r.url);
   359→    setFormDesc(r.description || "");
   360→    setFormCategory(r.category);
   361→    setFormTags(r.tags);
   362→    setDialogOpen(true);
   363→  };
   364→
   365→  const openAdd = () => {
   366→    setEditItem(null);
   367→    resetForm();
   368→    setDialogOpen(true);
   369→  };
   370→
   371→  const resetForm = () => {
   372→    setEditItem(null);
   373→    setFormTitle("");
   374→    setFormUrl("");
   375→    setFormDesc("");
   376→    setFormCategory("General");
   377→    setFormTags("");
   378→  };
   379→
   380→  const allCategories = ["All", ...categories];
   381→
   382→  return (
   383→    <div className="max-w-[1400px] mx-auto px-4 sm:px-6 py-6">
   384→      {/* TOP BAR */}
   385→      <div className="flex flex-col sm:flex-row items-start sm:items-center justify-between gap-4 mb-6">
   386→        <div>
   387→          <h2 className="text-2xl font-bold tracking-tight">Resource Library</h2>
   388→          <p className="text-sm text-muted-foreground mt-1">
   389→            {resources.length} resources in your collection
   390→          </p>
   391→        </div>
   392→        <Button onClick={openAdd} className="gap-2">
   393→          <Plus className="w-4 h-4" />
   394→          Add Resource
   395→        </Button>
   396→      </div>
   397→
   398→      {/* SEARCH & FILTERS */}
   399→      <div className="flex flex-col sm:flex-row gap-3 mb-6">
   400→        <div className="relative flex-1">
   401→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   402→          <Input
   403→            placeholder="Search resources..."
   404→            value={search}
   405→            onChange={(e) => setSearch(e.target.value)}
   406→            className="pl-9 h-10"
   407→          />
   408→        </div>
   409→        <div className="flex gap-2">
   410→          <Button
   411→            variant={showFavorites ? "default" : "outline"}
   412→            size="sm"
   413→            onClick={() => setShowFavorites(!showFavorites)}
   414→            className="gap-1.5 h-10"
   415→          >
   416→            <Heart className={`w-3.5 h-3.5 ${showFavorites ? "fill-current" : ""}`} />
   417→            Favorites
   418→          </Button>
   419→        </div>
   420→      </div>
   421→
   422→      {/* CATEGORY FILTERS */}
   423→      <div className="flex gap-2 mb-6 overflow-x-auto pb-1 scrollbar-none">
   424→        {allCategories.map((cat) => (
   425→          <button
   426→            key={cat}
   427→            onClick={() => setActiveCategory(cat)}
   428→            className={`inline-flex items-center gap-1.5 px-3 py-1.5 rounded-full text-sm font-medium whitespace-nowrap transition-colors ${
   429→              activeCategory === cat
   430→                ? "bg-foreground text-background"
   431→                : "bg-secondary text-secondary-foreground hover:bg-secondary/80"
   432→            }`}
   433→          >
   434→            {cat !== "All" && <span className="text-xs">{getCategoryIcon(cat)}</span>}
   435→            {cat}
   436→          </button>
   437→        ))}
   438→      </div>
   439→
   440→      {/* GRID */}
   441→      {loading ? (
   442→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   443→          {Array.from({ length: 6 }).map((_, i) => (
   444→            <Card key={i} className="overflow-hidden">
   445→              <CardContent className="p-5">
   446→                <div className="flex items-start gap-3">
   447→                  <Skeleton className="w-10 h-10 rounded-lg" />
   448→                  <div className="flex-1">
   449→                    <Skeleton className="h-5 w-24 mb-2" />
   450→                    <Skeleton className="h-3 w-full mb-1" />
   451→                    <Skeleton className="h-3 w-3/4" />
   452→                  </div>
   453→                </div>
   454→              </CardContent>
   455→            </Card>
   456→          ))}
   457→        </div>
   458→      ) : resources.length === 0 ? (
   459→        <div className="text-center py-20">
   460→          <FolderOpen className="w-12 h-12 mx-auto text-muted-foreground/50 mb-4" />
   461→          <h3 className="text-lg font-medium mb-1">No resources found</h3>
   462→          <p className="text-sm text-muted-foreground">
   463→            {showFavorites
   464→              ? "You haven't favorited any resources yet"
   465→              : "Try adjusting your search or filters"}
   466→          </p>
   467→        </div>
   468→      ) : (
   469→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   470→          {resources.map((r) => (
   471→            <ResourceCard
   472→              key={r.id}
   473→              resource={r}
   474→              onFavorite={() => toggleFavorite(r)}
   475→              onEdit={() => openEdit(r)}
   476→              onDelete={() => handleDelete(r.id)}
   477→            />
   478→          ))}
   479→        </div>
   480→      )}
   481→
   482→      {/* ADD/EDIT DIALOG */}
   483→      <Dialog open={dialogOpen} onOpenChange={(o) => { setDialogOpen(o); if (!o) resetForm(); }}>
   484→        <DialogContent className="sm:max-w-[480px]">
   485→          <DialogHeader>
   486→            <DialogTitle>{editItem ? "Edit Resource" : "Add Resource"}</DialogTitle>
   487→          </DialogHeader>
   488→          <div className="space-y-4 mt-2">
   489→            <div className="space-y-2">
   490→              <label className="text-sm font-medium">Title *</label>
   491→              <Input
   492→                placeholder="e.g. Figma"
   493→                value={formTitle}
   494→                onChange={(e) => setFormTitle(e.target.value)}
   495→              />
   496→            </div>
   497→            <div className="space-y-2">
   498→              <label className="text-sm font-medium">URL *</label>
   499→              <Input
   500→                placeholder="https://example.com"
   501→                value={formUrl}
   502→                onChange={(e) => setFormUrl(e.target.value)}
   503→              />
   504→            </div>
   505→            <div className="space-y-2">
   506→              <label className="text-sm font-medium">Description</label>
   507→              <Textarea
   508→                placeholder="Brief description of this resource..."
   509→                value={formDesc}
   510→                onChange={(e) => setFormDesc(e.target.value)}
   511→                rows={2}
   512→              />
   513→            </div>
   514→            <div className="space-y-2">
   515→              <label className="text-sm font-medium">Category</label>
   516→              <Select value={formCategory} onValueChange={setFormCategory}>
   517→                <SelectTrigger className="w-full">
   518→                  <SelectValue />
   519→                </SelectTrigger>
   520→                <SelectContent>
   521→                  {["Design", "Assets", "Development", "AI", "Productivity", "Reference", "Typography", "General"].map(
   522→                    (c) => (
   523→                      <SelectItem key={c} value={c}>
   524→                        {getCategoryIcon(c)} {c}
   525→                      </SelectItem>
   526→                    )
   527→                  )}
   528→                </SelectContent>
   529→              </Select>
   530→            </div>
   531→            <div className="space-y-2">
   532→              <label className="text-sm font-medium">Tags</label>
   533→              <Input
   534→                placeholder="design, ui, free (comma separated)"
   535→                value={formTags}
   536→                onChange={(e) => setFormTags(e.target.value)}
   537→              />
   538→            </div>
   539→            <div className="flex justify-end gap-2 pt-2">
   540→              <Button variant="outline" onClick={() => { setDialogOpen(false); resetForm(); }}>
   541→                Cancel
   542→              </Button>
   543→              <Button onClick={handleSubmit}>
   544→                {editItem ? "Save Changes" : "Add Resource"}
   545→              </Button>
   546→            </div>
   547→          </div>
   548→        </DialogContent>
   549→      </Dialog>
   550→    </div>
   551→  );
   552→}

   554→// ===================== RESOURCE CARD =====================
   555→
   556→function ResourceCard({
   557→  resource,
   558→  onFavorite,
   559→  onEdit,
   560→  onDelete,
   561→}: {
   562→  resource: Resource;
   563→  onFavorite: () => void;
   564→  onEdit: () => void;
   565→  onDelete: () => void;
   566→}) {
   567→  const [showMenu, setShowMenu] = useState(false);
   568→  const menuRef = useRef<HTMLDivElement>(null);
   569→  const hostname = (() => {
   570→    try { return new URL(resource.url).hostname; } catch { return resource.url; }
   571→  })();
   572→
   573→  useEffect(() => {
   574→    const handleClickOutside = (e: MouseEvent) => {
   575→      if (menuRef.current && !menuRef.current.contains(e.target as Node)) {
   576→        setShowMenu(false);
   577→      }
   578→    };
   579→    if (showMenu) document.addEventListener("mousedown", handleClickOutside);
   580→    return () => document.removeEventListener("mousedown", handleClickOutside);
   581→  }, [showMenu]);
   582→
   583→  return (
   584→    <motion.div
   585→      layout
   586→      initial={{ opacity: 0, y: 12 }}

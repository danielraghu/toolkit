   212→   212→function ResourcesSection() {
   213→   213→  const [resources, setResources] = useState<Resource[]>([]);
   214→   214→  const [categories, setCategories] = useState<string[]>([]);
   215→   215→  const [search, setSearch] = useState("");
   216→   216→  const [activeCategory, setActiveCategory] = useState("All");
   217→   217→  const [showFavorites, setShowFavorites] = useState(false);
   218→   218→  const [loading, setLoading] = useState(true);
   219→   219→  const [dialogOpen, setDialogOpen] = useState(false);
   220→   220→  const [editItem, setEditItem] = useState<Resource | null>(null);
   221→   221→  const [formTitle, setFormTitle] = useState("");
   222→   222→  const [formUrl, setFormUrl] = useState("");
   223→   223→  const [formDesc, setFormDesc] = useState("");
   224→   224→  const [formCategory, setFormCategory] = useState("General");
   225→   225→  const [formTags, setFormTags] = useState("");
   226→   226→
   227→   227→  const fetchResources = useCallback(async () => {
   228→   228→    try {
   229→   229→      setLoading(true);
   230→   230→      const params = new URLSearchParams();
   231→   231→      if (activeCategory !== "All") params.set("category", activeCategory);
   232→   232→      if (search) params.set("search", search);
   233→   233→      if (showFavorites) params.set("favorites", "true");
   234→   234→
   235→   235→      const res = await fetch(`/api/resources?${params}`);
   236→   236→      const data = await res.json();
   237→   237→      setResources(data.resources || []);
   238→   238→      setCategories(data.categories || []);
   239→   239→    } catch {
   240→   240→      toast.error("Failed to load resources");
   241→   241→    } finally {
   242→   242→      setLoading(false);
   243→   243→    }
   244→   244→  }, [activeCategory, search, showFavorites]);
   245→   245→
   246→   246→  useEffect(() => {
   247→   247→    fetchResources();
   248→   248→  }, [fetchResources]);
   249→   249→
   250→   250→  const handleSubmit = async () => {
   251→   251→    if (!formTitle.trim() || !formUrl.trim()) {
   252→   252→      toast.error("Title and URL are required");
   253→   253→      return;
   254→   254→    }
   255→   255→
   256→   256→    try {
   257→   257→      let url = formUrl.trim();
   258→   258→      if (!url.startsWith("http")) url = `https://${url}`;
   259→   259→
   260→   260→      const body = {
   261→   261→        title: formTitle.trim(),
   262→   262→        url,
   263→   263→        description: formDesc.trim(),
   264→   264→        category: formCategory,
   265→   265→        tags: formTags.trim(),
   266→   266→        faviconUrl: `https://www.google.com/s2/favicons?domain=${new URL(url).hostname}&sz=64`,
   267→   267→      };
   268→   268→
   269→   269→      if (editItem) {
   270→   270→        await fetch("/api/resources", {
   271→   271→          method: "PUT",
   272→   272→          headers: { "Content-Type": "application/json" },
   273→   273→          body: JSON.stringify({ id: editItem.id, ...body }),
   274→   274→        });
   275→   275→        toast.success("Resource updated");
   276→   276→      } else {
   277→   277→        await fetch("/api/resources", {
   278→   278→          method: "POST",
   279→   279→          headers: { "Content-Type": "application/json" },
   280→   280→          body: JSON.stringify(body),
   281→   281→        });
   282→   282→        toast.success("Resource added");
   283→   283→      }
   284→   284→
   285→   285→      setDialogOpen(false);
   286→   286→      resetForm();
   287→   287→      fetchResources();
   288→   288→    } catch {
   289→   289→      toast.error("Failed to save resource");
   290→   290→    }
   291→   291→  };
   292→   292→
   293→   293→  const handleDelete = async (id: string) => {
   294→   294→    try {
   295→   295→      await fetch(`/api/resources?id=${id}`, { method: "DELETE" });
   296→   296→      toast.success("Resource deleted");
   297→   297→      fetchResources();
   298→   298→    } catch {
   299→   299→      toast.error("Failed to delete");
   300→   300→    }
   301→   301→  };
   302→   302→
   303→   303→  const toggleFavorite = async (r: Resource) => {
   304→   304→    try {
   305→   305→      await fetch("/api/resources", {
   306→   306→        method: "PUT",
   307→   307→        headers: { "Content-Type": "application/json" },
   308→   308→        body: JSON.stringify({ id: r.id, isFavorite: !r.isFavorite }),
   309→   309→      });
   310→   310→      fetchResources();
   311→   311→    } catch {
   312→   312→      toast.error("Failed to update");
   313→   313→    }
   314→   314→  };
   315→   315→
   316→   316→  const openEdit = (r: Resource) => {
   317→   317→    setEditItem(r);
   318→   318→    setFormTitle(r.title);
   319→   319→    setFormUrl(r.url);
   320→   320→    setFormDesc(r.description || "");
   321→   321→    setFormCategory(r.category);
   322→   322→    setFormTags(r.tags);
   323→   323→    setDialogOpen(true);
   324→   324→  };
   325→   325→
   326→   326→  const openAdd = () => {
   327→   327→    setEditItem(null);
   328→   328→    resetForm();
   329→   329→    setDialogOpen(true);
   330→   330→  };
   331→   331→
   332→   332→  const resetForm = () => {
   333→   333→    setEditItem(null);
   334→   334→    setFormTitle("");
   335→   335→    setFormUrl("");
   336→   336→    setFormDesc("");
   337→   337→    setFormCategory("General");
   338→   338→    setFormTags("");
   339→   339→  };
   340→   340→
   341→   341→  const allCategories = ["All", ...categories];
   342→   342→
   343→   343→  return (
   344→   344→    <div className="max-w-[1200px] mx-auto px-6 py-8">
   345→   345→      {/* TOP BAR */}
   346→   346→      <div className="flex flex-col sm:flex-row items-start sm:items-center justify-between gap-4 mb-8">
   347→   347→        <div>
   348→   348→          <h2 className="text-[24px] font-bold tracking-tight text-white" style={{ lineHeight: 1.2 }}>Resource Library</h2>
   349→   349→          <p className="text-[14px] text-[#A0A0A0] mt-1" style={{ lineHeight: 1.5 }}>
   350→   350→            {resources.length} resources in your collection
   351→   351→          </p>
   352→   352→        </div>
   353→   353→        <button
   354→   354→          onClick={openAdd}
   355→   355→          className="inline-flex items-center gap-2 px-5 py-3 bg-[#FF6B35] hover:bg-[#FF5722] active:bg-[#E64A19] text-white rounded-lg text-[14px] font-semibold transition-all duration-200"
   356→   356→        >
   357→   357→          <Plus className="w-4 h-4" />
   358→   358→          Add Resource
   359→   359→        </button>
   360→   360→      </div>
   361→   361→
   362→   362→      {/* SEARCH & FILTERS */}
   363→   363→      <div className="flex flex-col sm:flex-row gap-3 mb-6">
   364→   364→        <div className="relative flex-1">
   365→   365→          <Search className="absolute left-4 top-1/2 -translate-y-1/2 w-5 h-5 text-[#606060]" />
   366→   366→          <input
   367→   367→            placeholder="Search resources..."
   368→   368→            value={search}
   369→   369→            onChange={(e) => setSearch(e.target.value)}
   370→   370→            className="w-full pl-12 pr-4 py-3 bg-[#2A2A2A] border border-[#333333] rounded-lg text-[14px] text-white placeholder:text-[#606060] outline-none focus:border-[#FF6B35] transition-colors duration-200"
   371→   371→          />
   372→   372→        </div>
   373→   373→        <button
   374→   374→          variant={showFavorites ? "default" : "outline"}
   375→   375→          onClick={() => setShowFavorites(!showFavorites)}
   376→   376→          className={`inline-flex items-center gap-1.5 px-4 py-3 rounded-lg text-[14px] font-medium transition-all duration-200 ${
   377→   377→            showFavorites
   378→   378→              ? "bg-[#FF6B35] text-white"
   379→   379→              : "bg-[#333333] text-[#A0A0A0] hover:bg-[#444444] hover:text-white"
   380→   380→          }`}
   381→   381→        >
   382→   382→          <Heart className={`w-4 h-4 ${showFavorites ? "fill-current" : ""}`} />
   383→   383→          Favorites
   384→   384→        </button>
   385→   385→      </div>
   386→   386→
   387→   387→      {/* CATEGORY FILTERS */}
   388→   388→      <div className="flex gap-2 mb-6 overflow-x-auto pb-1 scrollbar-none">
   389→   389→        {allCategories.map((cat) => (
   390→   390→          <button
   391→   391→            key={cat}
   392→   392→            onClick={() => setActiveCategory(cat)}
   393→   393→            className={`inline-flex items-center gap-1.5 px-4 py-2 rounded-full text-[12px] font-medium whitespace-nowrap transition-all duration-200 ${
   394→   394→              activeCategory === cat
   395→   395→                ? "bg-[#FF6B35] text-white"
   396→   396→                : "bg-[#333333] text-[#A0A0A0] hover:bg-[#444444] hover:text-white"
   397→   397→            }`}
   398→   398→          >
   399→   399→            {cat !== "All" && <span className="text-xs">{getCategoryIcon(cat)}</span>}
   400→   400→            {cat}
   401→   401→          </button>
   402→   402→        ))}
   403→   403→      </div>
   404→   404→
   405→   405→      {/* GRID */}
   406→   406→      {loading ? (
   407→   407→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-6">
   408→   408→          {Array.from({ length: 6 }).map((_, i) => (
   409→   409→            <div key={i} className="bg-[#1E1E1E] border border-[#333333] rounded-xl p-5">
   410→   410→              <div className="flex items-start gap-3">
   411→   411→                <Skeleton className="w-10 h-10 rounded-lg bg-[#333333]" />
   412→   412→                <div className="flex-1">
   413→   413→                  <Skeleton className="h-5 w-24 mb-2 bg-[#333333]" />
   414→   414→                  <Skeleton className="h-3 w-full mb-1 bg-[#333333]" />
   415→   415→                  <Skeleton className="h-3 w-3/4 bg-[#333333]" />
   416→   416→                </div>
   417→   417→              </div>
   418→   418→            </div>
   419→   419→          ))}
   420→   420→        </div>
   421→   421→      ) : resources.length === 0 ? (
   422→   422→        <div className="text-center py-20">
   423→   423→          <FolderOpen className="w-12 h-12 mx-auto text-[#606060] mb-4" />
   424→   424→          <h3 className="text-[18px] font-semibold text-white mb-1">No resources found</h3>
   425→   425→          <p className="text-[14px] text-[#A0A0A0]">
   426→   426→            {showFavorites
   427→   427→              ? "You haven't favorited any resources yet"
   428→   428→              : "Try adjusting your search or filters"}
   429→   429→          </p>
   430→   430→        </div>
   431→   431→      ) : (
   432→   432→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-6">
   433→   433→          {resources.map((r) => (
   434→   434→            <ResourceCard
   435→   435→              key={r.id}
   436→   436→              resource={r}
   437→   437→              onFavorite={() => toggleFavorite(r)}
   438→   438→              onEdit={() => openEdit(r)}
   439→   439→              onDelete={() => handleDelete(r.id)}
   440→   440→            />
   441→   441→          ))}
   442→   442→        </div>
   443→   443→      )}
   444→   444→
   445→   445→      {/* ADD/EDIT DIALOG */}
   446→   446→      <Dialog open={dialogOpen} onOpenChange={(o) => { setDialogOpen(o); if (!o) resetForm(); }}>
   447→   447→        <DialogContent className="sm:max-w-[480px] bg-[#1E1E1E] border-[#333333] rounded-xl">
   448→   448→          <DialogHeader>
   449→   449→            <DialogTitle className="text-white">{editItem ? "Edit Resource" : "Add Resource"}</DialogTitle>
   450→   450→          </DialogHeader>
   451→   451→          <div className="space-y-4 mt-2">
   452→   452→            <div className="space-y-2">
   453→   453→              <label className="text-[14px] font-medium text-[#A0A0A0]">Title *</label>
   454→   454→              <input
   455→   455→                placeholder="e.g. Figma"
   456→   456→                value={formTitle}
   457→   457→                onChange={(e) => setFormTitle(e.target.value)}
   458→   458→                className="w-full px-4 py-3 bg-[#2A2A2A] border border-[#333333] rounded-lg text-[14px] text-white placeholder:text-[#606060] outline-none focus:border-[#FF6B35] transition-colors duration-200"
   459→   459→              />
   460→   460→            </div>
   461→   461→            <div className="space-y-2">
   462→   462→              <label className="text-[14px] font-medium text-[#A0A0A0]">URL *</label>
   463→   463→              <input

   465→   465→                value={formUrl}
   466→   466→                onChange={(e) => setFormUrl(e.target.value)}
   467→   467→                className="w-full px-4 py-3 bg-[#2A2A2A] border border-[#333333] rounded-lg text-[14px] text-white placeholder:text-[#606060] outline-none focus:border-[#FF6B35] transition-colors duration-200"
   468→   468→              />
   469→   469→            </div>
